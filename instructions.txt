## Step 4: The connection pool

`src/config/db.js`:

```js
require("dotenv").config();
const { Pool } = require("pg");

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
});

module.exports = {
  query: (text, params) => pool.query(text, params),
  getClient: () => pool.connect(), // used for transactions
  pool,
};
```

A **pool** keeps a few open connections and reuses them, so you don't reconnect on every request. You'll import this one file everywhere you need the database.

## Step 5: The schema in SQL

`db/migrations/001_init.sql`:

```sql
-- Enums
CREATE TYPE user_role AS ENUM ('admin', 'teacher', 'student', 'parent');
CREATE TYPE payment_method AS ENUM ('mpesa_stk', 'mpesa_paybill', 'cash', 'bank');
CREATE TYPE payment_status AS ENUM ('pending', 'completed', 'failed', 'unallocated');
CREATE TYPE invoice_status AS ENUM ('unpaid', 'partial', 'paid');

-- Users (everyone who can log in)
CREATE TABLE users (
  id            SERIAL PRIMARY KEY,
  email         TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  role          user_role NOT NULL,
  is_active     BOOLEAN NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE teachers (
  id         SERIAL PRIMARY KEY,
  user_id    INTEGER NOT NULL UNIQUE REFERENCES users(id) ON DELETE CASCADE,
  staff_no   TEXT NOT NULL UNIQUE,
  first_name TEXT NOT NULL,
  last_name  TEXT NOT NULL,
  phone      TEXT
);

CREATE TABLE classes (
  id               SERIAL PRIMARY KEY,
  name             TEXT NOT NULL UNIQUE,
  level            TEXT,
  class_teacher_id INTEGER REFERENCES teachers(id) ON DELETE SET NULL
);

CREATE TABLE students (
  id             SERIAL PRIMARY KEY,
  user_id        INTEGER NOT NULL UNIQUE REFERENCES users(id) ON DELETE CASCADE,
  admission_no   TEXT NOT NULL UNIQUE,
  first_name     TEXT NOT NULL,
  last_name      TEXT NOT NULL,
  class_id       INTEGER NOT NULL REFERENCES classes(id),
  guardian_name  TEXT,
  guardian_phone TEXT,
  status         TEXT NOT NULL DEFAULT 'active',
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE academic_years (
  id         SERIAL PRIMARY KEY,
  name       TEXT NOT NULL UNIQUE,
  start_date DATE NOT NULL,
  end_date   DATE NOT NULL,
  is_current BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE terms (
  id               SERIAL PRIMARY KEY,
  academic_year_id INTEGER NOT NULL REFERENCES academic_years(id) ON DELETE CASCADE,
  name             TEXT NOT NULL,
  start_date       DATE NOT NULL,
  end_date         DATE NOT NULL,
  UNIQUE (academic_year_id, name)
);

-- What each class pays per term
CREATE TABLE fee_structures (
  id       SERIAL PRIMARY KEY,
  class_id INTEGER NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
  term_id  INTEGER NOT NULL REFERENCES terms(id) ON DELETE CASCADE,
  name     TEXT NOT NULL,
  amount   NUMERIC(12,2) NOT NULL CHECK (amount >= 0)
);

-- What a student owes for a term
CREATE TABLE invoices (
  id           SERIAL PRIMARY KEY,
  student_id   INTEGER NOT NULL REFERENCES students(id),
  term_id      INTEGER NOT NULL REFERENCES terms(id),
  total_amount NUMERIC(12,2) NOT NULL CHECK (total_amount >= 0),
  amount_paid  NUMERIC(12,2) NOT NULL DEFAULT 0 CHECK (amount_paid >= 0),
  status       invoice_status NOT NULL DEFAULT 'unpaid',
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (student_id, term_id)
);

CREATE TABLE invoice_items (
  id          SERIAL PRIMARY KEY,
  invoice_id  INTEGER NOT NULL REFERENCES invoices(id) ON DELETE CASCADE,
  description TEXT NOT NULL,
  amount      NUMERIC(12,2) NOT NULL
);

-- Every money-in event
CREATE TABLE payments (
  id            SERIAL PRIMARY KEY,
  student_id    INTEGER REFERENCES students(id),   -- nullable on purpose
  amount        NUMERIC(12,2) NOT NULL CHECK (amount > 0),
  method        payment_method NOT NULL,
  mpesa_receipt TEXT UNIQUE,                       -- blocks duplicate callbacks
  phone         TEXT,
  account_ref   TEXT,
  status        payment_status NOT NULL DEFAULT 'pending',
  raw_payload   JSONB,
  paid_at       TIMESTAMPTZ,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Which payment paid which invoice
CREATE TABLE payment_allocations (
  id         SERIAL PRIMARY KEY,
  payment_id INTEGER NOT NULL REFERENCES payments(id),
  invoice_id INTEGER NOT NULL REFERENCES invoices(id),
  amount     NUMERIC(12,2) NOT NULL CHECK (amount > 0)
);

-- Tracks each STK push you send
CREATE TABLE mpesa_requests (
  id                  SERIAL PRIMARY KEY,
  checkout_request_id TEXT NOT NULL UNIQUE,
  student_id          INTEGER NOT NULL REFERENCES students(id),
  amount              NUMERIC(12,2) NOT NULL,
  phone               TEXT NOT NULL,
  status              TEXT NOT NULL DEFAULT 'pending',
  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE audit_logs (
  id         SERIAL PRIMARY KEY,
  user_id    INTEGER REFERENCES users(id) ON DELETE SET NULL,
  action     TEXT NOT NULL,
  details    JSONB,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Indexes for the lookups you'll do constantly
CREATE INDEX idx_students_class      ON students(class_id);
CREATE INDEX idx_invoices_student    ON invoices(student_id);
CREATE INDEX idx_payments_student    ON payments(student_id);
CREATE INDEX idx_payments_status     ON payments(status);
CREATE INDEX idx_allocations_payment ON payment_allocations(payment_id);
CREATE INDEX idx_allocations_invoice ON payment_allocations(invoice_id);
```

### Why it's written this way

- **`NUMERIC(12,2)` for money, never `FLOAT`.** Floats cause rounding errors, which are unacceptable with fees.
- **`REFERENCES`** is a foreign key. Postgres refuses to save a student with a `class_id` that doesn't exist, so bad data can't get in.
- **`ON DELETE CASCADE`** means deleting a user also deletes their teacher/student profile. I left it off the money tables on purpose, so you can't wipe payment history by deleting a student.
- **`CHECK`** constraints reject negative amounts at the database level, even if your code has a bug.
- **`UNIQUE` on `mpesa_receipt`** is your protection against double-crediting. Postgres allows many NULLs in a unique column, so cash payments are fine.
- **`JSONB` for `raw_payload`** lets you store Safaricom's full response and still query inside it later.
- **No `balance` column.** Compute it as `total_amount - amount_paid`, so it can never drift out of sync.

## Step 6: The migration runner

`db/migrate.js` runs every `.sql` file in order and remembers which ones already ran:

```js
const fs = require("fs");
const path = require("path");
const { pool } = require("../src/config/db");

async function migrate() {
  await pool.query(`
    CREATE TABLE IF NOT EXISTS migrations (
      name       TEXT PRIMARY KEY,
      applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    )
  `);

  const dir = path.join(__dirname, "migrations");
  const files = fs.readdirSync(dir).filter((f) => f.endsWith(".sql")).sort();

  for (const file of files) {
    const { rows } = await pool.query("SELECT 1 FROM migrations WHERE name = $1", [file]);
    if (rows.length) continue; // already applied

    const sql = fs.readFileSync(path.join(dir, file), "utf8");
    const client = await pool.connect();
    try {
      await client.query("BEGIN");
      await client.query(sql);
      await client.query("INSERT INTO migrations (name) VALUES ($1)", [file]);
      await client.query("COMMIT");
      console.log("Applied:", file);
    } catch (err) {
      await client.query("ROLLBACK");
      console.error("Failed:", file, err.message);
      process.exit(1);
    } finally {
      client.release();
    }
  }

  await pool.end();
  console.log("Migrations done");
}

migrate();
```

Add this to `package.json` scripts:

```json
"scripts": {
  "migrate": "node db/migrate.js",
  "seed": "node db/seed.js"
}
```

Run it:

```bash
npm run migrate
```

**For future changes:** never edit `001_init.sql` after it has run. Create `002_add_receipts.sql` with your `ALTER TABLE` or `CREATE TABLE` statements, and run `npm run migrate` again. Only the new file runs. This is how real projects evolve their schema safely.

## Step 7: Seed the admin

`db/seed.js`:

```js
const bcrypt = require("bcrypt");
const { pool } = require("../src/config/db");

async function seed() {
  const hash = await bcrypt.hash("ChangeMe123!", 10);

  await pool.query(
    `INSERT INTO users (email, password_hash, role)
     VALUES ($1, $2, 'admin')
     ON CONFLICT (email) DO NOTHING`,
    ["admin@school.local", hash]
  );

  console.log("Admin ready");
  await pool.end();
}

seed();
```

```bash
npm run seed
```

Notice `$1, $2` instead of putting values into the string. These are **parameterized queries**. Never build SQL with template strings like `` `... '${email}'` ``, because that is how SQL injection happens. Always pass values in the array.

## A preview of the part that matters most: transactions

Allocating a payment means several changes that must all succeed or all fail together. Here is the pattern with `pg`, using the client from `getClient()`:

```js
const db = require("../config/db");

async function recordCashPayment(studentId, amount) {
  const client = await db.getClient();
  try {
    await client.query("BEGIN");

    const pay = await client.query(
      `INSERT INTO payments (student_id, amount, method, status, paid_at)
       VALUES ($1, $2, 'cash', 'completed', NOW()) RETURNING id`,
      [studentId, amount]
    );

    // ...allocate to invoices here...

    await client.query("COMMIT");
    return pay.rows[0].id;
  } catch (err) {
    await client.query("ROLLBACK"); // undo everything if anything fails
    throw err;
  } finally {
    client.release(); // always give the connection back
  }
}
```

If the server crashes between recording the payment and updating the invoice, `ROLLBACK` undoes the whole thing, so you never end up with money recorded but not allocated.
